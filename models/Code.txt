class_colors = {}
relation_colors = {
    "left of": (255, 0, 0),   # Blue
    "right of": (0, 255, 0),     # Green
    "near_to": (0, 0, 255),     # Red
    "inside": (255, 255, 0),  # Cyan
}

def get_class_color(class_name):
    """Assigns a unique and consistent color to each object class."""
    if class_name not in class_colors:
        class_colors[class_name] = (random.randint(50, 255), random.randint(50, 255), random.randint(50, 255))
    return class_colors[class_name]

# -------------------------------
# Function to Determine Spatial Relationships
# -------------------------------
def determine_spatial_relationship(box1, box2):
    """
    Determines the spatial relationship between two bounding boxes.
    Args:
        box1: Bounding box of the subject object (x1, y1, x2, y2).
        box2: Bounding box of the object object (x1, y1, x2, y2).
    Returns:
        A string describing the relationship (e.g., "left of", "right of", "near_to", "inside") or None.
    """
    x1_1, y1_1, x2_1, y2_1 = box1
    x1_2, y1_2, x2_2, y2_2 = box2

    center_x1, center_y1 = (x1_1 + x2_1) / 2, (y1_1 + y2_1) / 2
    center_x2, center_y2 = (x1_2 + x2_2) / 2, (y1_2 + y2_2) / 2

    width1, height1 = x2_1 - x1_1, y2_1 - y1_1
    width2, height2 = x2_2 - x1_2, y2_2 - y1_2

    # Check for "inside" relationship first as it's the most specific
    if x1_1 > x1_2 and y1_1 > y1_2 and x2_1 < x2_2 and y2_1 < y2_2:
        return "inside"

    # Horizontal relationships: "left of" / "right of"
    # Requires a significant vertical overlap to be considered horizontally aligned
    y_overlap = max(0, min(y2_1, y2_2) - max(y1_1, y1_2))
    # Using min(height1, height2) ensures even a small object next to a large one is counted
    if y_overlap / min(height1, height2) if min(height1, height2) > 0 else 0 > 0.4:
        if center_x1 < center_x2:
            return "left of"
        else:
            return "right of"

    # General proximity: "near_to"
    # This is a fallback for objects that are close but not perfectly aligned.
    distance = np.sqrt((center_x1 - center_x2)**2 + (center_y1 - center_y2)**2)
    # Threshold is based on the average size of the two objects
    nearness_threshold = ((width1 + width2 + height1 + height2) / 4) * 1.5
    if distance < nearness_threshold:
        return "near_to"

    return None

    # Store object info
    detected_objects.append({
        "name": unique_name,
        "class": class_name,
        "bbox": (x1, y1, x2, y2),
        "center": ((x1 + x2) // 2, (y1 + y2) // 2)
    })

    for j in range(len(detected_objects)):
        if i == j:
            continue

        subject = detected_objects[i]
        obj = detected_objects[j]

        relationship = determine_spatial_relationship(subject['bbox'], obj['bbox'])
        if relationship:
            scene_graph.append({
                "subject": subject,
                "predicate": relationship,
                "object": obj
            })

    obj = rel['object']

import math

def calculate_pixel_distances(detected_objects):
    """
    Calculates pixel distances between each pair of detected objects.
    
    Args:
        detected_objects (list): A list of detected objects where each object
                                 has a 'name' and 'center' (x, y).
    
    Returns:
        distances (dict): A dictionary where keys are tuples (obj1_name, obj2_name),
                          and values are the Euclidean pixel distances.
    """
    distances = {}
    for i in range(len(detected_objects)):
        for j in range(i + 1, len(detected_objects)):  # Avoid duplicate pairs
            obj1 = detected_objects[i]
            obj2 = detected_objects[j]
            
            # Extract centers
            x1, y1 = obj1["center"]
            x2, y2 = obj2["center"]
            
            # Euclidean distance
            dist = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            
            distances[(obj1["name"], obj2["name"])] = dist
    return distances


pixel_distances = calculate_pixel_distances(detected_objects)

print("\n--- PIXEL DISTANCES BETWEEN OBJECTS ---")
for pair, dist in pixel_distances.items():
    print(f"Distance between {pair[0]} and {pair[1]}: {dist:.2f} pixels")

def draw_pixel_distances(image, detected_objects):
    annotated_img = image.copy()
    
    for i in range(len(detected_objects)):
        for j in range(i + 1, len(detected_objects)):
            obj1 = detected_objects[i]
            obj2 = detected_objects[j]
            
            (x1, y1) = obj1["center"]
            (x2, y2) = obj2["center"]
            
            # Euclidean distance
            dist = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            
            # Midpoint for placing text
            mid_x, mid_y = (x1 + x2) // 2, (y1 + y2) // 2
            
            # Draw line between objects
            cv2.line(annotated_img, (x1, y1), (x2, y2), (0, 255, 0), 2)
            
            # Draw distance text
            cv2.putText(
                annotated_img,
                f"{dist:.1f}px",
                (mid_x, mid_y),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                (0, 0, 255),
                2
            )
    
    return annotated_img

import json
import math

def build_scene_json(detected_objects, scene_graph):
    """
    Build JSON with object detection details, distances, and spatial relationships.

    Args:
        detected_objects (list): List of detected objects with:
            - id (int)
            - name (str)      e.g., "car 1"
            - class (str)
            - bbox (tuple)    (x1, y1, x2, y2)
            - center (tuple)  (cx, cy)
            - confidence (float)
        scene_graph (list): List of relationships with subject, predicate, object.

    Returns:
        dict: JSON-compatible structure.
    """
    # -------------------------------
    # 1️⃣ Objects
    # -------------------------------
    objects = []
    for obj in detected_objects:
        objects.append({
            "id": obj["id"],                # numeric id (per class)
            #"name": obj["name"],            # e.g., "car 1"
            "class": obj["class"],
            "confidence": obj.get("confidence"),
            "bbox": obj["bbox"],            # (x1, y1, x2, y2)
            "center": obj["center"]         # (cx, cy)
        })

    # -------------------------------
    # 2️⃣ Distances (pixel distances between centers)
    # -------------------------------
    distances = []
    for i in range(len(detected_objects)):
        for j in range(i + 1, len(detected_objects)):
            o1 = detected_objects[i]
            o2 = detected_objects[j]

            (x1, y1) = o1["center"]
            (x2, y2) = o2["center"]
            dist = math.hypot(x1 - x2, y1 - y2)

            distances.append({
                "object1": o1["class"]+str(o1["id"]),
                "object2": o2["class"]+str(o2["id"]),
                "distance_px": round(dist, 2)
            })

    # -------------------------------
    # 3️⃣ Spatial relationships (from scene graph)
    # -------------------------------
    relationships = []
    for rel in scene_graph:
        relationships.append({
            "subject": rel.get("subject", {}).get("name"),
            "predicate": rel.get("predicate"),
            "object": rel.get("object", {}).get("name")
        })

    return {
        "objects": objects,
        "distances": distances,
        "relationships": relationships
    }


# -------------------------------
# Example usage after detection
# -------------------------------

# `detected_objects` is produced by your YOLO script (with "name" & "center" added)
# `scene_graph` is your list of relations (or [] if none)

scene_json = build_scene_json(detected_objects, scene_graph)
